# From client to Server
Go to http://localhost:3003/clientToServer.html
Then send a message from the client to the server

client
msg = 'hii user';
socket.emit('hello-event', msg);

Server
io.on('connection', (socket) => {
    socket.on('hello-event', (arg) => {
        console.log(`from client`, arg); // hi user
    })
})


# From Server to client
node clientToServer.js
Server
io.on('connection', () => {
    socket.emit('hello', 'world');
});

client
io.on('hello' (arg) => {
    console.log('receive', arg);
});
